import { join } from 'path';
import { config } from './config.js';
import { log } from './utils.js';

/**
 * Génère l'URL de démo locale pour un site
 * @param {string} slug
 * @returns {string}
 */
function generateLocalUrl(slug) {
  return `${config.deploy.baseUrl}/${slug}/index.html`;
}

/**
 * Prépare les métadonnées de déploiement pour un site
 * @param {Object} buildResult
 * @returns {Object}
 */
function prepareDeploymentInfo(buildResult) {
  const demoUrl = generateLocalUrl(buildResult.slug);

  return {
    ...buildResult,
    demoUrl,
    deployedAt: new Date().toISOString(),
    emailSubject: `Votre site vitrine démo est prêt !`,
    emailBody: generateEmailBody(buildResult, demoUrl),
  };
}

/**
 * Génère le corps d'email à envoyer au prospect
 * @param {Object} buildResult
 * @param {string} demoUrl
 * @returns {string}
 */
function generateEmailBody(buildResult, demoUrl) {
  return `Bonjour,

Nous avons créé une démonstration personnalisée de votre futur site vitrine.

🌐 Voir votre site démo : ${demoUrl}

Ce site a été entièrement personnalisé pour ${buildResult.lead} avec :
✅ Un design moderne et professionnel
✅ Votre contenu optimisé pour attirer des clients locaux
✅ Un formulaire de contact prêt à l'emploi

Intéressé(e) ? Répondez à cet email ou appelez-nous pour en savoir plus.

Cordialement,
L'équipe Auto-Demo`;
}

/**
 * Déploiement local (copie/organisation des fichiers)
 * @param {Object[]} buildResults
 * @returns {Promise<Object[]>}
 */
async function deployLocal(buildResults) {
  log('info', '[Deploy] Mode local — génération des URLs');

  return buildResults
    .filter(r => r.status === 'built')
    .map(result => {
      const deployInfo = prepareDeploymentInfo(result);
      log('success', `[Deploy] ${result.lead} → ${deployInfo.demoUrl}`);
      return deployInfo;
    });
}

/**
 * Point d'entrée principal du déploiement
 * @param {Object[]} buildResults - résultats du builder
 * @returns {Promise<Object[]>} informations de déploiement
 */
export async function deployAll(buildResults) {
  const successfulBuilds = buildResults.filter(r => r.status === 'built');

  if (successfulBuilds.length === 0) {
    log('warn', '[Deploy] Aucun site à déployer');
    return [];
  }

  log('info', `[Deploy] Déploiement de ${successfulBuilds.length} site(s) en mode "${config.deploy.mode}"`);

  switch (config.deploy.mode) {
    case 'local':
      return deployLocal(buildResults);

    default:
      log('warn', `[Deploy] Mode "${config.deploy.mode}" non reconnu, fallback local`);
      return deployLocal(buildResults);
  }
}

/**
 * Affiche le rapport de déploiement dans la console
 * @param {Object[]} deployResults
 */
export function printDeployReport(deployResults) {
  if (deployResults.length === 0) return;

  console.log('\n' + '═'.repeat(60));
  console.log('📊 RAPPORT DE DÉPLOIEMENT');
  console.log('═'.repeat(60));

  deployResults.forEach((result, index) => {
    console.log(`\n${index + 1}. ${result.lead}`);
    console.log(`   📧 Email : ${result.email || 'N/A'}`);
    console.log(`   🌐 URL   : ${result.demoUrl}`);
    console.log(`   📁 Fichier: ${result.outputPath}`);
  });

  console.log('\n' + '═'.repeat(60));
  console.log(`✅ ${deployResults.length} site(s) générés avec succès`);
  console.log('═'.repeat(60) + '\n');
}
